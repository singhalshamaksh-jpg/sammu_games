+async function liveWebSearch(query) {
+    let snippets = [];
+    try {
+        const wRes = await fetch(`https://wikipedia.org{encodeURIComponent(query)}&format=json&srlimit=2`);
+        const wData = await wRes.json();
+        for (const h of (wData?.query?.search || [])) {
+            snippets.push(`${h.title}: ${h.snippet.replace(/<[^>]+>/g, '').replace(/&quot;/g, '"').replace(/&#039;/g, "'")}`);
+        }
+    } catch (e) {}
+    try {
+        const dRes = await fetch(`https://duckduckgo.com{encodeURIComponent(query)}&format=json&no_html=1`);
+        const dData = await dRes.json();
+        if (dData.AbstractText) snippets.unshift(dData.AbstractText);
+    } catch (e) {}
+    return snippets.join(" | ").slice(0, 650);
